Output strictly as JSON with keys 'image_prompt' and 'interview_script'. """ response = client.chat.completions.create( model="gpt-4o-mini", messages=[{"role": "user", "content": prompt}], response_format={"type": "json_object"}, ) data = json.loads(response.choices[0].message.content) return data["image_prompt"], data["interview_script"] def generate_character_image(image_prompt, role): """Generates a portrait image using DALL-E 3.""" print(" Generating character avatar image...") full_prompt = f"A realistic front-facing centered portrait photo of a character, actor style interview setup, lighting cinematic, {role} archetype: {image_prompt}" response = client.images.generate( model="dall-e-3", prompt=full_prompt, size="1024x1024", quality="standard", n=1, ) image_url = response.data[0].url return image_url def create_talking_video(image_url, script_text): """Uses D-ID API to generate a talking head video from the image and text script.""" print(" Submitting job to generate talking video...") url = "https://api.d-id.com/talks" headers = { "Authorization": f"Basic {D_ID_API_KEY}", "Content-Type": "application/json", } payload = { "script": { "type": "text", "subtitles": "false", "provider": { "type": "microsoft", "voice_id": "en-US-GuyNeural", # You can adjust voice models here }, "input": script_text, }, "config": {"fluent": "false", "pad_audio": "0.0"}, "source_url": image_url, } response = requests.post(url, json=payload, headers=headers) talk_data = response.json() if "id" not in talk_data: print("Error creating video job:", talk_data) return None talk_id = talk_data["id"] print(f" Video job created ID: {talk_id}.
